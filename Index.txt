<script>
(function () {
  const $ = (id) => document.getElementById(id);
  const fields = ['arriveDate','arriveTime','loadDate','loadTime','pickupName','pickupContact','pickupPhone',
    'dropName','dropPhone','dropAddress','fare','vehicle','weight','item','note','requester'];
  const REQUIRED_FIELDS = [
    { id: 'pickupName', label: '상차지 상호' },
    { id: 'dropName', label: '하차지 상호/성함' },
    { id: 'dropAddress', label: '하차지 주소' },
    { id: 'requester', label: '요청자' }
  ];

  function todayStr() { return new Date().toISOString().slice(0,10); }
  function setDefaultDates() {
    $('arriveDate').value = todayStr();
    $('loadDate').value = todayStr();
  }
  function formatMD(dateStr) {
    if (!dateStr) return '';
    const [y, m, d] = dateStr.split('-');
    return `${parseInt(m,10)}/${parseInt(d,10)}`;
  }
  function formatTimeKR(timeStr) {
    if (!timeStr) return '';
    let [h, m] = timeStr.split(':').map(Number);
    const period = h < 12 ? '오전' : '오후';
    let h12 = h % 12; if (h12 === 0) h12 = 12;
    return m === 0 ? `${period} ${h12}시` : `${period} ${h12}시 ${m}분`;
  }

  function buildParts() {
    const arriveText = `${formatMD($('arriveDate').value)} ${formatTimeKR($('arriveTime').value)}착 요청!`;
    const loadText = `${formatMD($('loadDate').value)} ${formatTimeKR($('loadTime').value)}부터 상차가능`;
    return { arriveText, loadText };
  }

  function buildMessage() {
    const { arriveText, loadText } = buildParts();
    const pickupName = $('pickupName').value.trim() || '(상차지 미입력)';
    const pickupContact = $('pickupContact').value.trim();
    const pickupPhone = $('pickupPhone').value.trim();
    const dropName = $('dropName').value.trim() || '(하차지 미입력)';
    const dropPhone = $('dropPhone').value.trim();
    const dropAddress = $('dropAddress').value.trim();
    const fare = $('fare').value.trim();
    const vehicle = $('vehicle').value.trim();
    const weight = $('weight').value.trim();
    const item = $('item').value.trim();
    const note = $('note').value.trim();

    const lines = [];
    lines.push('배차요청드립니다.');
    lines.push('');
    lines.push(`일정 : ${arriveText}`);
    lines.push(loadText);
    lines.push('');
    lines.push(`상차지 : ${pickupName}`);
    lines.push(`담당자 : ${pickupContact}${pickupPhone ? ' (' + pickupPhone + ')' : ''}`);
    lines.push('');
    lines.push(`하차지 : ${dropName}${dropPhone ? ' (' + dropPhone + ')' : ''}`);
    lines.push(`주소 : ${dropAddress}`);
    lines.push('');
    lines.push(`운임 : ${fare}`);
    lines.push(`차량 : ${vehicle}`);
    lines.push(`중량 : ${weight}`);
    lines.push(`품목 : ${item}`);
    lines.push('');
    lines.push(`비고: ${note}`);
    return lines.join('\n');
  }

  function render() { $('output').textContent = buildMessage(); }

  function clearInvalidStyles() {
    REQUIRED_FIELDS.forEach(f => {
      const wrap = $('field-' + f.id);
      if (wrap) wrap.classList.remove('invalid');
    });
    $('validationMsg').style.display = 'none';
  }

  // 누락된 필수 항목이 있으면 표시하고 true(문제있음)를 반환
  function checkRequired() {
    clearInvalidStyles();
    const missing = [];
    REQUIRED_FIELDS.forEach(f => {
      const val = $(f.id).value.trim();
      if (!val) {
        missing.push(f.label);
        const wrap = $('field-' + f.id);
        if (wrap) wrap.classList.add('invalid');
      }
    });
    if (missing.length > 0) {
      const msg = $('validationMsg');
      msg.textContent = '다음 항목을 채워주세요: ' + missing.join(', ');
      msg.style.display = 'block';
      return true;
    }
    return false;
  }

  function saveLocalDefaults() {
    try {
      if ($('savePickup').checked) {
        localStorage.setItem('quickDispatchDefaults', JSON.stringify({
          pickupName: $('pickupName').value, pickupContact: $('pickupContact').value,
          pickupPhone: $('pickupPhone').value, fare: $('fare').value, vehicle: $('vehicle').value,
          requester: $('requester').value, savePickup: true
        }));
      } else {
        localStorage.removeItem('quickDispatchDefaults');
      }
    } catch (e) {}
  }
  function loadLocalDefaults() {
    try {
      const raw = localStorage.getItem('quickDispatchDefaults');
      if (!raw) return;
      const data = JSON.parse(raw);
      ['pickupName','pickupContact','pickupPhone','fare','vehicle','requester'].forEach(k => {
        if (data[k]) $(k).value = data[k];
      });
      if (data.savePickup) $('savePickup').checked = true;
    } catch (e) {}
  }

  fields.forEach(id => $(id).addEventListener('input', () => { render(); saveLocalDefaults(); }));
  $('requester').addEventListener('change', () => { render(); saveLocalDefaults(); });
  $('savePickup').addEventListener('change', saveLocalDefaults);

  $('copyBtn').addEventListener('click', () => {
    if (checkRequired()) return;
    const text = buildMessage();
    const btn = $('copyBtn');
    const done = () => {
      btn.textContent = '복사됨 ✓'; btn.classList.add('copied');
      setTimeout(() => { btn.textContent = '문구 복사하기'; btn.classList.remove('copied'); }, 1500);
    };
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(text).then(done).catch(() => fallbackCopy(text, done));
    } else { fallbackCopy(text, done); }
  });
  function fallbackCopy(text, cb) {
    const ta = document.createElement('textarea');
    ta.value = text; ta.style.position = 'fixed'; ta.style.left = '-9999px';
    document.body.appendChild(ta); ta.select();
    try { document.execCommand('copy'); cb(); } catch (e) {}
    document.body.removeChild(ta);
  }

  function resetAfterSave() {
    // 화물 정보 + 하차지는 비우고, 저장해둔 상차지 기본값은 유지
    ['dropName','dropPhone','dropAddress','fare','vehicle','weight','item'].forEach(id => { $(id).value = ''; });
    $('note').value = '출발전 착지 연락부탁드립니다.';
    setDefaultDates();
    if (!$('savePickup').checked) {
      $('pickupName').value = ''; $('pickupContact').value = ''; $('pickupPhone').value = '';
    }
    render();
  }

  $('saveBtn').addEventListener('click', () => {
    if (checkRequired()) return;
    const { arriveText, loadText } = buildParts();
    const data = {
      requester: $('requester').value.trim(),
      arriveText, loadText,
      pickupName: $('pickupName').value.trim(),
      pickupContact: $('pickupContact').value.trim(),
      pickupPhone: $('pickupPhone').value.trim(),
      dropName: $('dropName').value.trim(),
      dropPhone: $('dropPhone').value.trim(),
      dropAddress: $('dropAddress').value.trim(),
      fare: $('fare').value.trim(),
      vehicle: $('vehicle').value.trim(),
      weight: $('weight').value.trim(),
      item: $('item').value.trim(),
      note: $('note').value.trim(),
      message: buildMessage()
    };
    const btn = $('saveBtn');
    const status = $('status');
    btn.disabled = true; btn.textContent = '저장 중...';
    status.textContent = ''; status.className = 'status';

    google.script.run
      .withSuccessHandler(() => {
        btn.disabled = false; btn.textContent = '스프레드시트에 저장'; btn.classList.add('saved');
        setTimeout(() => btn.classList.remove('saved'), 1200);
        status.textContent = '저장되었습니다.'; status.className = 'status ok';
        resetAfterSave();
        loadRecent();
      })
      .withFailureHandler((err) => {
        btn.disabled = false; btn.textContent = '스프레드시트에 저장';
        status.textContent = '저장 실패: ' + (err && err.message ? err.message : err);
        status.className = 'status err';
      })
      .saveDispatch(data);
  });

  $('resetBtn').addEventListener('click', () => {
    fields.forEach(id => { if (id !== 'arriveDate' && id !== 'loadDate') $(id).value = ''; });
    $('note').value = '출발전 착지 연락부탁드립니다.';
    setDefaultDates(); render(); $('status').textContent = ''; clearInvalidStyles();
  });

  function renderRecent(entries) {
    const list = $('recentList');
    list.innerHTML = '';
    if (!entries || entries.length === 0) {
      list.innerHTML = '<li class="recent-empty">아직 저장된 요청이 없어요.</li>';
      return;
    }
    entries.forEach(e => {
      const li = document.createElement('li');
      li.innerHTML = '<span class="recent-time">' + e.timestamp + '</span>' +
        (e.requester ? e.requester + ' · ' : '') + e.pickupName + ' → ' + e.dropName;
      list.appendChild(li);
    });
  }

  function loadRecent() {
    google.script.run
      .withSuccessHandler(renderRecent)
      .withFailureHandler(() => { $('recentList').innerHTML = '<li class="recent-empty">불러오지 못했어요.</li>'; })
      .getRecentEntries(5);
  }

  setDefaultDates(); loadLocalDefaults(); render(); loadRecent();
})();
</script>
